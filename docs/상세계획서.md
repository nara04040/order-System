
# 맥도날드 주문 시스템 함수형 프로그래밍 구현 계획서

## 1. 프로젝트 개요
JavaScript 함수형 프로그래밍을 활용하여 맥도날드 주문 시스템의 핵심 로직 구현

## 2. 기술 스택
- 순수 JavaScript (ES6+)
- 함수형 프로그래밍 패러다임

## 3. 핵심 원칙
- 불변성(Immutability)
- 순수 함수(Pure Functions)
- 함수 합성(Function Composition)
- 부작용 최소화(Minimizing Side Effects)

## 4. 모듈 구조

>src/
> ├── data.js             # 메뉴 데이터와 시스템 상태 관리
> ├── utils.js            # 함수형 프로그래밍 유틸리티 함수
> ├── menuFunctions.js    # 메뉴 조회 및 필터링 함수
> ├── orderFunctions.js   # 주문 생성 및 수정 함수
> ├── processingFunctions.js # 주문 처리 및 상태 변경 함수
> ├── eventSystem.js      # 이벤트 구독/발행 시스템
> └── main.js             # 메인 애플리케이션 로직 및 테스트

### 4.2 유틸리티 함수 (`utils.js`)

```javascript
// 함수 합성 유틸리티
const pipe = (...fns) => x => fns.reduce((acc, fn) => fn(acc), x);
const compose = (...fns) => x => fns.reduceRight((acc, fn) => fn(acc), x);

// 불변 객체 업데이트 유틸리티
const updateObject = (obj, updates) => ({ ...obj, ...updates });
const updateArray = (arr, index, newItem) => [
  ...arr.slice(0, index),
  newItem,
  ...arr.slice(index + 1)
];

export { pipe, compose, updateObject, updateArray };
```

### 4.4 주문 관련 함수 (`orderFunctions.js`)
```javascript
// 새 주문 생성
const createOrder = (items = []) => ({
  items: items,
  total: calculateTotal(items),
  status: 'creating',
  timestamp: Date.now()
});

// 주문에 아이템 추가
const addItemToOrder = (order, item) => {
  const newItems = [...order.items, item];
  return {
    ...order,
    items: newItems,
    total: calculateTotal(newItems)
  };
};

// 주문 총액 계산
const calculateTotal = (items) => 
  items.reduce((sum, item) => sum + item.price, 0);

// 주문번호 생성 
const generateOrderNumber = (lastOrderNumber) => lastOrderNumber + 1;

// 주문 접수 처리
const placeOrder = (state, order) => {
  const orderNumber = generateOrderNumber(state.lastOrderNumber);
  const finalizedOrder = {
    ...order,
    orderNumber,
    status: 'placed'
  };
  
  return {
    ...state,
    orders: [...state.orders, finalizedOrder],
    orderQueue: [...state.orderQueue, finalizedOrder],
    currentOrder: null,
    lastOrderNumber: orderNumber
  };
};

export { 
  createOrder, 
  addItemToOrder, 
  calculateTotal, 
  generateOrderNumber, 
  placeOrder 
};
```

### 4.5 주문 처리 함수 (`processingFunctions.js`)
```javascript
// 주문 상태 업데이트
const updateOrderStatus = (order, newStatus) => ({
  ...order,
  status: newStatus
});

// 큐에서 다음 주문 처리 시작
const processNextOrder = (state) => {
  // 큐가 비어있으면 상태 그대로 반환
  if (state.orderQueue.length === 0) return state;
  
  const [nextOrder, ...remainingQueue] = state.orderQueue;
  const processingOrder = updateOrderStatus(nextOrder, 'processing');
  
  // 처리 중인 주문 업데이트
  const updatedOrders = state.orders.map(order => 
    order.orderNumber === processingOrder.orderNumber ? processingOrder : order
  );
  
  return {
    ...state,
    orders: updatedOrders,
    orderQueue: remainingQueue
  };
};

// 주문 처리 완료
const completeOrder = (state, orderNumber) => {
  const completedOrders = state.orders.map(order => 
    order.orderNumber === orderNumber 
      ? updateOrderStatus(order, 'completed') 
      : order
  );
  
  return {
    ...state,
    orders: completedOrders
  };
};

export { updateOrderStatus, processNextOrder, completeOrder };
```

### 4.6 이벤트 시스템 (`eventSystem.js`)
```javascript
// 단순 이벤트 시스템
const createEventSystem = () => {
  const listeners = {};
  
  // 이벤트 구독
  const subscribe = (event, callback) => {
    if (!listeners[event]) listeners[event] = [];
    listeners[event].push(callback);
    
    // 구독 취소 함수 반환
    return () => {
      listeners[event] = listeners[event].filter(cb => cb !== callback);
    };
  };
  
  // 이벤트 발생
  const emit = (event, data) => {
    if (!listeners[event]) return;
    listeners[event].forEach(callback => callback(data));
  };
  
  return { subscribe, emit };
};

export { createEventSystem };
```

## 5. 단계별 동작 설명

### 5.1 시스템 초기화
1. 필요한 모듈 임포트
2. 초기 상태 설정 (`initialState`)
3. 이벤트 시스템 초기화
   - 이벤트 구독 설정 (주문 접수, 주문 처리, 주문 완료 등)

### 5.2 주문 생성 단계
1. 빈 주문 객체 생성 (`createOrder()`)
2. 메뉴에서 아이템 선택 (`findMenuItem(menu, itemId)`)
3. 주문에 아이템 추가 (`addItemToOrder(order, item)`)
   - 불변성 유지: 원본 주문은 변경되지 않고 새 주문 객체 반환
   - 선택한 아이템 목록 업데이트
   - 주문 총액 자동 계산
4. 필요에 따라 2-3단계 반복

### 5.3 주문 접수 단계
1. 주문 완료 결정 (사용자 확인)
2. 주문 접수 처리 (`placeOrder(state, order)`)
   - 고유 주문번호 생성
   - 주문 상태를 'placed'로 설정
   - 전체 주문 목록과 처리 대기열에 추가
   - 상태 불변성 유지: 새로운 상태 객체 반환

### 5.4 주문 처리 단계
1. 주문 큐에서 다음 주문 가져오기 (`processNextOrder(state)`)
   - 큐의 첫 번째 주문을 처리 상태로 변경
   - 주문 상태를 'processing'으로 업데이트
   - 큐에서 해당 주문 제거
   - 불변성 유지: 새 상태 객체 반환
2. 주문 처리 진행 (실제 조리 과정)

### 5.5 주문 완료 단계
1. 주문 완료 처리 (`completeOrder(state, orderNumber)`)
   - 해당 주문의 상태를 'completed'로 업데이트
   - 불변성 유지: 새 상태 객체 반환
2. 주문 완료 이벤트 발생 (`events.emit('order:completed', order)`)
   - 구독자들에게 주문 완료 알림 (화면 표시, 소리 재생 등)

## 6. 구현 단계
1. 기본 데이터 구조 및 유틸리티 함수 구현
2. 메뉴 관련 기능 구현 및 테스트
3. 주문 생성 및 수정 기능 구현
4. 주문 처리 파이프라인 구현
5. 이벤트 시스템 구현
6. 통합 및 전체 흐름 테스트

## 7. 테스트 시나리오 예시

```javascript
// main.js - 시스템 동작 예시

import { menu, initialState } from './data.js';
import { getMenu, findMenuItem } from './menuFunctions.js';
import { createOrder, addItemToOrder, placeOrder } from './orderFunctions.js';
import { processNextOrder, completeOrder } from './processingFunctions.js';
import { createEventSystem } from './eventSystem.js';

// 이벤트 시스템 생성
const events = createEventSystem();

// 이벤트 구독 설정
events.subscribe('order:placed', (order) => {
  console.log(`주문 ${order.orderNumber}번이 접수되었습니다!`);
  console.log(`주문 내역: ${order.items.map(item => item.name).join(', ')}`);
  console.log(`총액: ${order.total}원`);
});

events.subscribe('order:processing', (order) => {
  console.log(`주문 ${order.orderNumber}번 조리 시작!`);
});

events.subscribe('order:completed', (order) => {
  console.log(`주문 ${order.orderNumber}번이 완료되었습니다!`);
  console.log(`픽업 카운터에서 음식을 받아가세요.`);
});

// 상태 관리 함수
let state = initialState;

// 상태 업데이트 함수 (순수 함수의 결과를 적용)
const updateState = (newState) => {
  state = newState;
  return state;
};

// 전체 주문 흐름 시뮬레이션
const simulateOrderFlow = () => {
  // 1. 메뉴 아이템 선택
  const bigMac = findMenuItem(menu, 'b1');
  const fries = findMenuItem(menu, 's1');
  const coke = findMenuItem(menu, 'd1');
  
  // 2. 주문 생성 및 아이템 추가
  let order = createOrder();
  order = addItemToOrder(order, bigMac);
  order = addItemToOrder(order, fries);
  order = addItemToOrder(order, coke);
  
  console.log("=== 주문 생성 ===");
  console.log(order);
  
  // 3. 주문 접수
  console.log("\n=== 주문 접수 ===");
  state = updateState(placeOrder(state, order));
  events.emit('order:placed', state.orders[0]);
  
  // 4. 주문 처리 시작
  console.log("\n=== 주문 처리 시작 ===");
  state = updateState(processNextOrder(state));
  events.emit('order:processing', state.orders[0]);
  
  // 5. 주문 완료
  console.log("\n=== 주문 완료 ===");
  state = updateState(completeOrder(state, state.orders[0].orderNumber));
  events.emit('order:completed', state.orders[0]);
  
  return state;
};

// 시뮬레이션 실행
const finalState = simulateOrderFlow();
console.log("\n=== 최종 시스템 상태 ===");
console.log(finalState);
```

## 8. 함수형 프로그래밍 적용 포인트

- **불변성 유지**: 모든 함수는 원본 데이터를 수정하지 않고 새 객체 반환
- **순수 함수**: 같은 입력에 항상 같은 출력 반환, 부작용 없음
- **함수 합성**: 작은 기능의 함수들로 복잡한 동작 구현
- **고차 함수**: map, filter, reduce 등으로 데이터 처리
- **상태 관리**: 모든 상태 변경은 순수 함수를 통해 새 상태로 변환
- **이벤트 기반 구조**: 구독/발행 패턴으로 시스템 컴포넌트 간 결합도 낮춤
